Tasks:

1. npm init -y
npm install express
npm install --save-dev @types/express

2. touch index.js
const express = require('express');
const app = express();

const PORT = 8080;

app.get('/health', (req, res) => {
    res.status(200).json({ status: "OK" });
});

app.listen(PORT, () => {
    console.log(`Server running on port ${PORT}`);
});

To run the server: 
node index.js
curl http://localhost:8080/health
